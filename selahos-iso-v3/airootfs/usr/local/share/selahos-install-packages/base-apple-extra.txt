# SelahSeedCore can recommend linux-lts for older Mac GPUs (hardware-db.json
# "kernel": "linux-lts" entries, e.g. Broadwell/iMac16,2) -- it must actually
# be installed or the later grub-set-default call has no entry to point at
# and silently falls back to linux-zen, which doesn't have the same track
# record on that hardware.
linux-lts
linux-lts-headers
